    \begin{frame}{Versiebeheer en software}
		\begin{itemize}

            \begin{itemize}

		\end{itemize}

    \begin{frame}{Branch -- Fetch -- Pull -- Push}
            \begin{exampleblock}{Terminologie}
				\begin{itemize}
					\item Branch: alternatieve versie van de repository.
					\item \texttt{main}: hoofdbranch, bevat stabiele versie van project.
					\item Fetch: check GitHub voor nieuwe branches en commits, nog geen lokale wijzigingen.
					\item Pull en Push: afzonderlijk voor elke branch.
				\end{itemize}
			\end{exampleblock}
			\begin{alertblock}{Good Practice}
				\begin{itemize}
				\item Zet \texttt{main} niet lokaal
				\item Werk altijd in secundaire branch
				\end{itemize}
			\end{alertblock}
	\end{frame}

\item Bronbranch: waarvan wijzigingen worden overgenomen.
\item Doelbranch: waarin wijzigingen worden opgenomen.
